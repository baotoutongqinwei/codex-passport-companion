cmake_minimum_required(VERSION 3.16)
project(companion_ui_preview C CXX ASM)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/managed_components/lvgl__lvgl" lvgl)
add_executable(companion_ui_preview preview.c "${REPO}/main/companion_model.c" "${REPO}/main/companion_device_model.c" "${REPO}/assets/fonts/passport_font_16.c")
target_include_directories(companion_ui_preview PRIVATE stubs "${REPO}/main" "${REPO}/components/bsp/include")
target_link_libraries(companion_ui_preview PRIVATE lvgl m)
add_executable(offline_ui_preview offline_preview.c "${REPO}/main/companion_offline_model.c" "${REPO}/main/companion_battery_log_model.c" "${REPO}/main/companion_device_model.c" "${REPO}/assets/fonts/passport_font_16.c")
target_include_directories(offline_ui_preview PRIVATE stubs "${REPO}/main" "${REPO}/components/bsp/include")
target_link_libraries(offline_ui_preview PRIVATE lvgl m)
